// SHARED ENGINE — this block is byte-identical in Noa.zBSH (monitor) and
// Noa.AVWAP (strategy). If you edit one, edit both, or the thing you watch
// stops being the thing you tested.
// ═══════════════════════════════════════════════════════════════════════════

grpC = "Core (Z entry)"
src    = input.source(close, "Z source", group = grpC)
length = input.int(21, "VWAP / Z lookback", minval = 2, maxval = 500, group = grpC)
gain   = input.float(100.0, "Smoother gain", minval = 0.1, maxval = 4000.0, step = 10,
     group = grpC,
     tooltip = "Drives the alpha-beta smoother below. Capped at 4000 on purpose: " +
               "the position gain is sqrt(gain/5000), which reaches 1.0 at " +
               "gain = 5000 and above that the filter overshoots every " +
               "observation and rings.")

grpZ = "Z bands"
topUpper = input.float(2.5,  "Top upper",    group = grpZ)
upperZ   = input.float(1.5,  "Upper",        group = grpZ)
lowerZ   = input.float(-1.5, "Lower",        group = grpZ)
botLower = input.float(-2.5, "Bottom lower", group = grpZ)

grpF = "Flip zone"
zFlip     = input.float(0.0, "Flip center", group = grpF)
flipRange = input.float(0.5, "Flip half-width", minval = 0.0, step = 0.1, group = grpF)
flipUpper = zFlip + flipRange
flipLower = zFlip - flipRange

grpP = "Entry policy"
policy = input.string("Flip Cross", "Policy",
     options = ["Flip Cross", "Mean Revert"], group = grpP,
     tooltip = "Measured over 11.6 years of NIFTY and RELIANCE 15m/30m:\n" +
               "  Flip Cross  +2.4 bp gross per trade\n" +
               "  Mean Revert −0.5 to −3.1 bp gross per trade\n" +
               "Mean Revert loses money BEFORE costs, on both symbols and both " +
               "timeframes. It is kept only so you can reproduce that.")
useSmoothTrig = input.bool(false, "Trigger on the smoother instead of raw Z", group = grpP)
tradeLong     = input.bool(true,  "Allow BUY",  group = grpP)
tradeShort    = input.bool(true,  "Allow SELL", group = grpP)

grpX = "Exit"
exitRule = input.string("Session only", "Exit rule",
     options = ["Session only", "AVWAP Cross", "AVWAP beyond", "Flip band"],
     group = grpX,
     tooltip = "Gross bp per trade, NIFTY 15m, 11.6y, one bar of execution lag:\n" +
               "  Session only  4.23  (2,401 trades, −30% max DD)\n" +
               "  AVWAP Cross   2.35  (4,832 trades, −67% max DD)\n" +
               "  AVWAP beyond  2.14  (5,302 trades, −72% max DD)\n" +
               "  Flip band     0.85  (6,576 trades, −91% max DD)\n\n" +
               "The flip-band exit was the original default and is the worst of " +
               "the four by a wide margin: a Flip Cross entry lands you INSIDE " +
               "the flip band by construction, so it exits on the next bar. " +
               "Median hold was 1 bar and 66% of trades lasted exactly one.")
avwapSrc = input.source(hlc3, "AVWAP price source", group = grpX)
minBarsInTrade = input.int(1, "Min bars before an AVWAP exit", minval = 0, group = grpX)

grpS = "Session (NSE)"
useSession   = input.bool(true, "Restrict to session", group = grpS)
sessStr      = input.session("0915-1530", "Session", group = grpS)
tz           = input.string("Asia/Kolkata", "Timezone", group = grpS)
entryStartH  = input.int(9,  "Earliest entry hour",   minval = 0, maxval = 23, group = grpS)
entryStartM  = input.int(20, "Earliest entry minute", minval = 0, maxval = 59, group = grpS)
flatBeforeClose = input.bool(true, "Force flat before session end", group = grpS)
flatH = input.int(15, "Flatten hour",   minval = 0, maxval = 23, group = grpS)
flatM = input.int(20, "Flatten minute", minval = 0, maxval = 59, group = grpS)

// ── Z: price against a rolling volume-weighted mean ────────────────────────
// NSE index volume is 0 on every bar (measured: 212,626 of 212,626 on NIFTY),
// so src*volume / volume is 0/0 and the original engine produced na on any
// spot index — silently, with no plot and no warning. Fall back to an equal
// weighted mean when there is no volume to weight with.
float volRaw   = nz(volume, 0.0)
bool  hasVol   = ta.sma(volRaw, length) > 0
float wt       = hasVol ? volRaw : 1.0

float vwMean = ta.sma(src * wt, length) / ta.sma(wt, length)
// The dispersion is weighted with the same weights as the mean. The original
// took an unweighted second moment around a weighted first moment, which is
// internally inconsistent — sigma then belongs to a different distribution
// than the mean z is measured from.
float vwVar  = ta.sma(math.pow(src - vwMean, 2) * wt, length) / ta.sma(wt, length)
float vwSd   = math.sqrt(vwVar)
float zRaw   = (not na(vwSd) and vwSd > 0) ? (src - vwMean) / vwSd : na

// ── Alpha-beta (g-h) smoother ──────────────────────────────────────────────
// Named honestly: this is NOT a Kalman filter. There is no state covariance and
// no innovation variance, so the gains are fixed by input rather than derived
// from the noise, and there is no sqrt(Q) to standardise anything by. It is a
// constant-velocity smoother with hand-set gains — legitimate, just not Kalman.
//   alpha = sqrt(gain/5000)   position gain
//   beta  = gain/10000        velocity gain
// Both corrections must read the SAME previous state; updating the position
// first and then deriving the velocity from the new value uses the wrong
// innovation and double-counts it.
float gAlpha = math.sqrt(gain / 5000.0)
float gBeta  = gain / 10000.0

var float kf   = na
var float velo = 0.0
float prevKf   = kf
float prevVelo = nz(velo, 0.0)
float kfBase   = na(prevKf) ? zRaw : prevKf
float dk       = zRaw - kfBase
if not na(dk)
    velo := prevVelo + gBeta * dk
    kf   := kfBase + dk * gAlpha + velo

float zTrig = useSmoothTrig ? kf : zRaw

// ── Session gates ──────────────────────────────────────────────────────────
bool inSession = not useSession or not na(time(timeframe.period, sessStr, tz))
int  tod       = hour(time, tz) * 60 + minute(time, tz)
bool timeOk    = not useSession or tod >= entryStartH * 60 + entryStartM
bool forceFlat = useSession and flatBeforeClose and tod >= flatH * 60 + flatM

// ── Entry triggers — ta.* must be called on every bar ──────────────────────
bool xFlipUp = ta.crossover(zTrig, zFlip)
bool xFlipDn = ta.crossunder(zTrig, zFlip)
bool xBotUp  = ta.crossover(zTrig, botLower)
bool xLowUp  = ta.crossover(zTrig, lowerZ)
bool xTopDn  = ta.crossunder(zTrig, topUpper)
bool xUppDn  = ta.crossunder(zTrig, upperZ)

bool rawBuy  = policy == "Flip Cross" ? xFlipUp : (xBotUp or xLowUp)
bool rawSell = policy == "Flip Cross" ? xFlipDn : (xTopDn or xUppDn)

bool buyTrig  = inSession and timeOk and not forceFlat and tradeLong  and rawBuy
bool sellTrig = inSession and timeOk and not forceFlat and tradeShort and rawSell

// ── Position + anchored VWAP from the signal bar ───────────────────────────
var int   state     = 0
var int   barsInPos = 0
var float sumPV     = 0.0
var float sumV      = 0.0
var float avwap     = na

float aWt = hasVol ? volRaw : 1.0
float aPx = avwapSrc

bool toBuy  = false
bool toSell = false
bool toHold = false
string exitWhy = ""

if state != 0
    sumPV += aPx * aWt
    sumV  += aWt
    avwap := sumV > 0 ? sumPV / sumV : na
    barsInPos += 1

bool xuAvwap = ta.crossunder(close, avwap)
bool xoAvwap = ta.crossover(close, avwap)
bool enoughBars  = barsInPos > minBarsInTrade
bool beyondLong  = not na(avwap) and close < avwap
bool beyondShort = not na(avwap) and close > avwap
bool inFlipNow   = not na(zTrig) and zTrig >= flipLower and zTrig <= flipUpper

bool avwapExitLong  = state == 1  and enoughBars and not na(avwap) and (exitRule == "AVWAP Cross" ? xuAvwap : exitRule == "AVWAP beyond" ? beyondLong  : false)
bool avwapExitShort = state == -1 and enoughBars and not na(avwap) and (exitRule == "AVWAP Cross" ? xoAvwap : exitRule == "AVWAP beyond" ? beyondShort : false)
bool flipExit       = exitRule == "Flip band" and state != 0 and inFlipNow

bool wantHold = forceFlat or avwapExitLong or avwapExitShort or flipExit
if wantHold and state != 0
    exitWhy   := forceFlat ? "SESSION" : flipExit ? "FLIP" : "AVWAP"
    state     := 0
    barsInPos := 0
    sumPV     := 0.0
    sumV      := 0.0
    avwap     := na
    toHold    := true

if not toHold
    if buyTrig and state != 1
        state     := 1
        barsInPos := 0
        sumPV     := aPx * aWt
        sumV      := aWt
        avwap     := sumV > 0 ? sumPV / sumV : na
        toBuy     := true
    else if sellTrig and state != -1
        state     := -1
        barsInPos := 0
        sumPV     := aPx * aWt
        sumV      := aWt
        avwap     := sumV > 0 ? sumPV / sumV : na
        toSell    := true
